def _http_json(
    method: str,
    url: str,
    *,
    headers: Optional[Dict[str, str]] = None,
    body: Optional[Dict[str, Any]] = None,
    timeout: float = 5.0,
) -> Tuple[int, Dict[str, Any]]:
    data: Optional[bytes]
    req_headers = {"Accept": "application/json"}
    if headers:
        req_headers.update(headers)
    if body is None:
        data = None
    else:
        raw = json.dumps(body).encode("utf-8")
        data = raw
        req_headers.setdefault("Content-Type", "application/json")

    req = urllib.request.Request(url, data=data, method=method, headers=req_headers)
    try:
        with urllib.request.urlopen(req, timeout=timeout) as resp:
            status = int(resp.status)
            payload_bytes = resp.read()
    except urllib.error.HTTPError as e:
        status = int(e.code)
        payload_bytes = e.read() or b""
    except Exception as e:
        raise RuntimeError(f"request_failed {method} {url}: {e}") from e

    if payload_bytes:
        try:
            payload = json.loads(payload_bytes.decode("utf-8"))
        except Exception:
            payload = {"_raw": payload_bytes.decode("utf-8", errors="replace")}
    else:
        payload = {}

    return status, payload


def _wait_until_ready(base_url: str, proc: subprocess.Popen[bytes], timeout_s: float = 8.0) -> None:

class AuthRepository:
    def __init__(self, conn: sqlite3.Connection):
        self._conn = conn

    def create_user(
        self, username: str, salt: bytes, password_hash: bytes, created_at: float
    ) -> int:
        try:
            cur = self._conn.execute(
                "INSERT INTO users(username, salt, password_hash, created_at) VALUES (?, ?, ?, ?)",
                (username, salt, password_hash, created_at),
            )
        except sqlite3.IntegrityError as exc:
            raise DuplicateUsernameError(username) from exc
        self._conn.commit()
        if cur.lastrowid is None:
            raise RuntimeError("Failed to create user")
        return int(cur.lastrowid)

    def get_user_credentials(self, username: str) -> tuple[int, bytes, bytes] | None:
        cur = self._conn.execute(
            "SELECT id, salt, password_hash FROM users WHERE username = ?", (username,)
        )
        row = cur.fetchone()
        if row is None:
            return None
        return int(row["id"]), bytes(row["salt"]), bytes(row["password_hash"])

    def store_token(self, token: str, user_id: int, created_at: float) -> None:
        self._conn.execute(
            "INSERT INTO tokens(token, user_id, created_at) VALUES (?, ?, ?)",
            (token, user_id, created_at),
        )
        self._conn.commit()

    def get_user_id_for_token(self, token: str) -> int | None:
        cur = self._conn.execute("SELECT user_id FROM tokens WHERE token = ?", (token,))

        if row is None:

        return int(row["user_id"])

class AuthService:
    def __init__(self, repo: AuthRepository, now: Callable[[], float] | None = None):
        self._repo = repo
        self._now = now or time.time

    def register(self, username: str, password: str) -> int:
        salt = secrets.token_bytes(16)
        pw_hash = _hash_password(password=password, salt=salt)
        return self._repo.create_user(
            username=username, salt=salt, password_hash=pw_hash, created_at=self._now()
        )

    def login(self, username: str, password: str) -> str | None:
        creds = self._repo.get_user_credentials(username)
        if creds is None:
            return None
        user_id, salt, expected = creds
        actual = _hash_password(password=password, salt=salt)
        if not hmac.compare_digest(expected, actual):
            return None
        token = secrets.token_urlsafe(32)
        self._repo.store_token(token=token, user_id=user_id, created_at=self._now())
        return token

    def authenticate_token(self, token: str) -> int | None:
        return self._repo.get_user_id_for_token(token)

class PaymentsService:
    def __init__(self, payments_repo: PaymentsRepository, orders_repo: OrdersRepository):

        self, *, user_id: int, order_id: int, idempotency_key: str
    ) -> tuple[str, int | None]:
        owner = self._orders.get_order_owner(order_id)
        if owner is None:
            return "not_found", None
        if owner != user_id:
            return "forbidden", None

            payment_id, existing_order_id = existing

                return "idempotency_conflict", None
            return "paid", payment_id

        now = time.time()

        )

            self._orders.set_order_status(order_id=order_id, status="paid")
        return "paid", payment_id

def _read_json(handler: BaseHTTPRequestHandler) -> dict[str, Any]:
    length_raw = handler.headers.get("Content-Length")

        raise ApiError(HTTPStatus.BAD_REQUEST, "missing_body", "Body required")
    try:

def _json_response(
    handler: BaseHTTPRequestHandler, status: HTTPStatus, body: Mapping[str, Any]
) -> None:

    handler.send_response(int(status))
    handler.send_header("Content-Type", "application/json")
    handler.send_header("Content-Length", str(len(payload)))
    handler.end_headers()
    handler.wfile.write(payload)

def _require_bearer(headers: Mapping[str, str], auth: AuthService) -> int:
    authz = headers.get("Authorization")
    if authz is None or not authz.startswith("Bearer "):
        raise ApiError(HTTPStatus.UNAUTHORIZED, "missing_token", "Bearer token required")
    token = authz.removeprefix("Bearer ").strip()
    user_id = auth.authenticate_token(token)
    if user_id is None:
        raise ApiError(HTTPStatus.UNAUTHORIZED, "invalid_token", "Invalid token")
    return user_id

                HTTPStatus.CONFLICT,
                "idempotency_conflict",
                "Idempotency-Key already used for a different order",
            )
        _json_response(self, HTTPStatus.OK, {"status": status, "id": payment_id})
