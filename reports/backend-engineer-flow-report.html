Run Flow Report

Intent: Build the BACKEND of a URL shortener with analytics — `shortlink` — as an HTTP service in app.py (Python standard library ONLY — http.server + sqlite3 + json + secrets, no pip installs, no frontend). It starts with `python app.py` on the PORT env var (default 8000) and DB_PATH env var (default shortlink.db), and applies a SQL schema via a small MIGRATIONS mechanism on startup (an ordered list of migration statements recorded in a schema_migrations table so re-runs are idempotent). Use ThreadingHTTPServer with a per-request connection (concurrency-safe). Endpoints: - GET /health -> 200 'ok' - POST /links -> body {"url": "<http(s) url>"}; validates the URL (else 400 JSON error); mints a COLLISION-SAFE short code (base62, generated with the `secrets` module, retried on the unique constraint) and returns 201 {"code", "short_url", "url"} - GET /{code} -> 301 redirect to the original URL (Location header), AND records a click event (timestamp, referrer from the Referer header, user-agent from the User-Agent header); unknown code -> 404 - GET /api/links/{code}/stats -> 200 JSON with total_clicks, top_referrers (referrer -> count, descending), and clicks_by_day (an ISO date -> count map, time-bucketed by UTC day) Put the base62 codec + the data-access/migrations logic in a separate module and the HTTP layer in app.py. Data MUST survive a process restart. Author the tests independently: unit tests for the base62 codec + code-uniqueness under concurrency + the day-bucketed aggregation against a known event fixture, and prove the running service end-to-end (mint -> redirect -> stats, and that a minted code still redirects after a restart).

Stdlib only. This module owns:
- base62 codec
- schema migrations
- sqlite operations (insert link, resolve, click tracking, stats)

The HTTP layer lives in app.py.
"""

from __future__ import annotations

import datetime as _dt
import secrets
import sqlite3
from typing import Any, Dict, Iterable, List, Optional, Tuple


BASE62_ALPHABET = "0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz"
BASE62_INDEX = {c: i for i, c in enumerate(BASE62_ALPHABET)}


def base62_encode(num: int) -> str:
    if num < 0:
        raise ValueError("num must be non-negative")
    if num == 0:
        return BASE62_ALPHABET[0]
    out: List[str] = []
    while num:
        num, rem = divmod(num, 62)
        out.append(BASE62_ALPHABET[rem])
    return "".join(reversed(out))


def base62_decode(s: str) -> int:
    if not s:
        raise ValueError("s must be non-empty")
    num = 0
    for ch in s:
        try:
            val = BASE62_INDEX[ch]
        except KeyError as e:
            raise ValueError("invalid base62") from e
        num = num * 62 + val
    return num


MIGRATIONS: List[Tuple[str, str]] = [
    (
        "0001_init",
        """
        CREATE TABLE IF NOT EXISTS schema_migrations (
            name TEXT PRIMARY KEY,
            applied_at TEXT NOT NULL
        );

        CREATE TABLE IF NOT EXISTS links (
            code TEXT PRIMARY KEY,
            url TEXT NOT NULL,
            created_at TEXT NOT NULL
        );

        CREATE TABLE IF NOT EXISTS clicks (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            code TEXT NOT NULL,
            clicked_at TEXT NOT NULL,
            referrer TEXT,
            user_agent TEXT,
            FOREIGN KEY(code) REFERENCES links(code)
        );

        CREATE INDEX IF NOT EXISTS idx_clicks_code_clicked_at ON clicks(code, clicked_at);
        """,
    )
]


def connect(db_path: str) -> sqlite3.Connection:
    conn = sqlite3.connect(db_path, timeout=30, isolation_level=None)
    conn.row_factory = sqlite3.Row
    conn.execute("PRAGMA journal_mode=WAL;")
    conn.execute("PRAGMA foreign_keys=ON;")
    return conn


def apply_migrations(conn: sqlite3.Connection) -> None:
    conn.execute(
        "CREATE TABLE IF NOT EXISTS schema_migrations (name TEXT PRIMARY KEY, applied_at TEXT NOT NULL);"
    )
    applied = {row["name"] for row in conn.execute("SELECT name FROM schema_migrations").fetchall()}

    for name, sql in MIGRATIONS:
        if name in applied:
            continue
        with conn:
            conn.executescript(sql)
            conn.execute(
                "INSERT INTO schema_migrations(name, applied_at) VALUES(?, ?)",
                (name, utc_now().isoformat(timespec="seconds") + "Z"),
            )


def utc_now() -> _dt.datetime:
    return _dt.datetime.now(tz=_dt.timezone.utc)


def _mint_code(length: int = 8) -> str:
    # 62^8 ~ 2e14; token safe and uniform-ish.
    # Use random bytes -> int -> base62, then pad/trim.
    nbits = length * 6  # log2(62)~5.95
    nbytes = (nbits + 7) // 8
    num = int.from_bytes(secrets.token_bytes(nbytes), "big")
    code = base62_encode(num)
    if len(code) < length:
        code = code.rjust(length, BASE62_ALPHABET[0])
    elif len(code) > length:
        code = code[-length:]
    return code

import shortlink_store as store


def _json_bytes(obj: Any) -> bytes:
    return (json.dumps(obj, separators=(",", ":")) + "\n").encode("utf-8")


def _send_json(handler: BaseHTTPRequestHandler, status: int, obj: Any) -> None:
    body = _json_bytes(obj)
    handler.send_response(status)
    handler.send_header("Content-Type", "application/json; charset=utf-8")
    handler.send_header("Content-Length", str(len(body)))
    handler.end_headers()
    handler.wfile.write(body)


def _read_json(handler: BaseHTTPRequestHandler) -> Tuple[Optional[Dict[str, Any]], Optional[str]]:
    length = handler.headers.get("Content-Length")
    if length is None:
        return None, "missing Content-Length"
    try:
        n = int(length)
    except ValueError:
        return None, "invalid Content-Length"
    raw = handler.rfile.read(n)
    try:
        data = json.loads(raw.decode("utf-8"))
    except Exception:
        return None, "invalid JSON"
    if not isinstance(data, dict):
        return None, "JSON body must be an object"
    return data, None


def _is_valid_http_url(url: str) -> bool:
    try:
        p = urlparse(url)
    except Exception:
        return False
    if p.scheme not in ("http", "https"):
        return False
    if not p.netloc:
        return False
    return True


_CODE_RE = re.compile(r"^[0-9A-Za-z]{1,64}$")


class Handler(BaseHTTPRequestHandler):
    server_version = "shortlink/1.0"

    def _db(self):
        return store.connect(self.server.db_path)  # type: ignore[attr-defined]

    def do_GET(self) -> None:  # noqa: N802
        path = self.path.split("?", 1)[0]

        if path == "/health":
            body = b"ok"
            self.send_response(200)
            self.send_header("Content-Type", "text/plain; charset=utf-8")
            self.send_header("Content-Length", str(len(body)))
            self.end_headers()
            self.wfile.write(body)
            return

        if path.startswith("/api/links/") and path.endswith("/stats"):
            parts = path.split("/")
            if len(parts) != 5:
                _send_json(self, 404, {"error": "not found"})
                return
            code = parts[3]
            if not _CODE_RE.match(code):
                _send_json(self, 404, {"error": "not found"})
                return
            conn = self._db()
            try:
                stats = store.get_stats(conn, code)
            finally:
                conn.close()
            if stats is None:
                _send_json(self, 404, {"error": "not found"})
                return
            _send_json(self, 200, stats)
            return

        if path.startswith("/") and len(path) > 1 and "/" not in path[1:]:
            code = path[1:]
            if not _CODE_RE.match(code):
                self.send_response(404)
                self.end_headers()
                return
            conn = self._db()

def _pick_free_port() -> int:
    with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as s:
        s.bind(("127.0.0.1", 0))
        return int(s.getsockname()[1])


def _wait_http_ok(url: str, timeout_s: float = 10.0) -> None:
    deadline = time.time() + timeout_s
    last_err = None
    while time.time() < deadline:
        try:
            with urlopen(url, timeout=1) as resp:
                if resp.status == 200:
                    return
        except Exception as e:  # pragma: no cover
            last_err = e
        time.sleep(0.1)
    raise AssertionError(f"service did not become ready: {last_err}")

class TestBase62Codec(unittest.TestCase):

        # include boundary-ish values and some larger ints

    def test_base62_encode_rejects_negative(self):
        with self.assertRaises(ValueError):

class TestCreateLinkConcurrency(unittest.TestCase):
    def test_create_link_concurrent_uniqueness(self):
        # Uses a real sqlite file so multiple connections/threads contend on the same DB.
        with tempfile.NamedTemporaryFile(suffix=".db") as tf:
            db_path = tf.name

            conn = store.connect(db_path)
            try:
                store.apply_migrations(conn)
            finally:
                conn.close()

            n_threads = 25
            codes = []
            codes_lock = threading.Lock()
            errors = []

            def worker(i: int) -> None:
                try:
                    c = store.connect(db_path)
                    try:
                        code = store.create_link(c, f"https://example.com/{i}")
                    finally:
                        c.close()
                    with codes_lock:
                        codes.append(code)
                except Exception as e:  # pragma: no cover (diagnostics)
                    with codes_lock:
                        errors.append(e)

            threads = [threading.Thread(target=worker, args=(i,)) for i in range(n_threads)]
            for t in threads:
                t.start()
            for t in threads:
                t.join(timeout=10)

            self.assertFalse(errors, f"unexpected errors: {errors}")
            self.assertEqual(len(codes), n_threads)
            self.assertEqual(len(set(codes)), n_threads, "codes must be unique under concurrency")
