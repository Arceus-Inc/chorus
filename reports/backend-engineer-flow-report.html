import contextlib

from dataclasses import dataclass
from typing import Any, Dict, Optional, Tuple


def free_port() -> int:
    with contextlib.closing(socket.socket(socket.AF_INET, socket.SOCK_STREAM)) as s:
        s.bind(("127.0.0.1", 0))
        return int(s.getsockname()[1])


def http_json(
    method: str,
    url: str,
    body: Optional[Dict[str, Any]] = None,
    headers: Optional[Dict[str, str]] = None,
    timeout: float = 5.0,
) -> Tuple[int, Dict[str, Any], Dict[str, str]]:
    data = None
    req_headers = {"Accept": "application/json"}
    if headers:
        req_headers.update(headers)
    if body is not None:
        data = json.dumps(body).encode("utf-8")
        req_headers.setdefault("Content-Type", "application/json")
    req = urllib.request.Request(url, data=data, headers=req_headers, method=method)
    try:
        with urllib.request.urlopen(req, timeout=timeout) as resp:
            raw = resp.read()
            txt = raw.decode("utf-8") if raw else ""
            parsed: Dict[str, Any] = {}
            if txt:
                parsed = json.loads(txt)
            return resp.status, parsed, dict(resp.headers)
    except urllib.error.HTTPError as e:
        raw = e.read()
        txt = raw.decode("utf-8") if raw else ""
        parsed: Dict[str, Any] = {}
        if txt:
            try:
                parsed = json.loads(txt)
            except json.JSONDecodeError:
                parsed = {"_raw": txt}
        return int(e.code), parsed, dict(e.headers)


@dataclass
class Check:
    name: str
    ok: bool
    detail: Any


def main() -> int:
    checks: list[Check] = []
    evidence: Dict[str, Any] = {}

    port = free_port()
    base = f"http://127.0.0.1:{port}"

    with tempfile.TemporaryDirectory() as td:
        db_path = os.path.join(td, "test.sqlite3")
        env = os.environ.copy()
        env["PORT"] = str(port)
        env["DB_PATH"] = db_path

        cmd = [sys.executable, "main.py"]
        evidence["boot"] = {"cmd": cmd, "port": port, "db_path": db_path}

        log_path = os.path.join(td, "server.log")
        proc: Optional[subprocess.Popen[bytes]] = None

        try:

                proc = subprocess.Popen(
                    cmd,
                    env=env,
                    stdout=log,
                    stderr=subprocess.STDOUT,

def connect(db_path: str) -> sqlite3.Connection:
    conn = sqlite3.connect(db_path, check_same_thread=False)
    conn.row_factory = sqlite3.Row
    apply_migrations(conn)
    return conn

def apply_migrations(conn: sqlite3.Connection) -> None:
    conn.execute("PRAGMA foreign_keys = ON;")
    conn.executescript(
        """
        CREATE TABLE IF NOT EXISTS schema_migrations (
            version INTEGER PRIMARY KEY
        );
        """
    )
    applied_versions = {
        row[0] for row in conn.execute("SELECT version FROM schema_migrations")
    }
    for migration in MIGRATIONS:
        if migration.version in applied_versions:
            continue
        conn.executescript(migration.sql)
        conn.execute(
            "INSERT INTO schema_migrations(version) VALUES (?)", (migration.version,)
        )

    ) -> User:
        try:
            cur = self._conn.execute(
                "INSERT INTO users(username, password_salt, password_hash) VALUES (?,?,?)",
                (username, password_salt, password_hash),
            )
        except sqlite3.IntegrityError as e:
            raise UserAlreadyExistsError(username) from e
        self._conn.commit()
        if cur.lastrowid is None:
            raise RuntimeError("failed to create user")
        return User(id=int(cur.lastrowid), username=username)

        row = self._conn.execute(
            "SELECT id, username, password_salt, password_hash FROM users WHERE username = ?",
            (username,),
        ).fetchone()
        if row is None:
            raise InvalidCredentialsError("unknown user")
        return (
            User(id=int(row["id"]), username=str(row["username"])),
            bytes(row["password_salt"]),
            bytes(row["password_hash"]),
        )

    def store_token(self, *, token: str, user_id: int) -> None:
        created_at = datetime.now(tz=timezone.utc).isoformat()
        self._conn.execute(
            "INSERT INTO tokens(token, user_id, created_at) VALUES (?,?,?)",
            (token, user_id, created_at),
        )
        self._conn.commit()

    def user_id_from_token(self, *, token: str) -> int:
        row = self._conn.execute(
            "SELECT user_id FROM tokens WHERE token = ?",
            (token,),
        ).fetchone()
        if row is None:
            raise TokenNotFoundError()
        return int(row["user_id"])

def generate_salt() -> bytes:

def hash_password(password: str, *, salt: bytes) -> bytes:
    return hashlib.pbkdf2_hmac("sha256", password.encode("utf-8"), salt, 200_000)


def verify_password(password: str, *, salt: bytes, password_hash: bytes) -> bool:
    candidate = hash_password(password, salt=salt)
    return hmac.compare_digest(candidate, password_hash)

@dataclass(frozen=True)
class RouteMatch:
    params: dict[str, str]


def match(template: str, path: str) -> RouteMatch | None:
    tpl_parts = [p for p in template.strip("/").split("/") if p]
    path_parts = [p for p in path.strip("/").split("/") if p]
    if len(tpl_parts) != len(path_parts):
        return None
    params: dict[str, str] = {}
    for t, p in zip(tpl_parts, path_parts, strict=True):
        if t.startswith("{") and t.endswith("}"):
            params[t[1:-1]] = p
        elif t != p:
            return None
    return RouteMatch(params=params)

        self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(body)))

        length = int(self.headers.get("Content-Length", "0"))
        raw = self.rfile.read(length) if length else b"{}"

        except json.JSONDecodeError as e:
            raise ValueError("invalid json") from e
        if not isinstance(data, dict):
            raise ValueError("json must be object")

def _free_port() -> int:
    with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as s:
        s.bind(("127.0.0.1", 0))
        return int(s.getsockname()[1])


def _wait_http_ready(
    host: str, port: int, proc: subprocess.Popen[str], timeout_s: float = 5.0
) -> None:
    deadline = time.time() + timeout_s
    last_exc: OSError | None = None
    while time.time() < deadline:
        if proc.poll() is not None:
            raise RuntimeError("server process exited early")
        try:
            conn = HTTPConnection(host, port, timeout=0.2)
            conn.request("GET", "/")
            conn.getresponse().read()
            return
        except OSError as e:
            last_exc = e
            time.sleep(0.05)
        finally:
            with contextlib.suppress(OSError):
                conn.close()
    raise RuntimeError(f"server not ready within timeout; last error: {last_exc!r}")

def _http_json(
    base_url: str,
    method: str,
    path: str,
    *,
    json_body: dict[str, object] | None = None,
    headers: dict[str, str] | None = None,
) -> tuple[int, dict[str, object] | None, str]:
    assert base_url.startswith("http://")
    hostport = base_url.removeprefix("http://")
    host, port_s = hostport.split(":")
    conn = HTTPConnection(host, int(port_s), timeout=2)
    body = None
    send_headers = {"Accept": "application/json"}
    if headers:
        send_headers.update(headers)
    if json_body is not None:
        body = json.dumps(json_body).encode("utf-8")
        send_headers["Content-Type"] = "application/json"
        send_headers["Content-Length"] = str(len(body))
    conn.request(method, path, body=body, headers=send_headers)
    resp = conn.getresponse()
    raw = resp.read().decode("utf-8")
    data: object | None = None
    if raw:
        with contextlib.suppress(json.JSONDecodeError):
            data = json.loads(raw)
    return resp.status, data if isinstance(data, dict) else None, raw
