Intent: Build the BACKEND of a small commerce API as an HTTP service (Python standard library ONLY — http.server + sqlite3 + json + secrets + hashlib, no pip installs, no frontend). It starts with `python main.py` on the PORT env var (default 8000) and DB_PATH env var (default commerce.db), and applies a SQL schema via a small MIGRATIONS mechanism on startup (an ordered list recorded in a schema_migrations table so re-runs are idempotent). Use ThreadingHTTPServer with a per-request connection. There are THREE DOMAINS, each its own bounded context — organise the code as ONE PACKAGE PER DOMAIN (`auth/`, `orders/`, `payments/`), NOT by file-type folders, and point dependencies inward (transport/HTTP -> service -> data-access -> domain): - GET /health -> 200 'ok' auth: - POST /auth/register {"email","password"} -> 201; store a SALTED password HASH, never the plaintext (hashlib); duplicate email -> 409. - POST /auth/login {"email","password"} -> 200 {"token"} (an opaque bearer token minted with `secrets`); wrong credentials -> 401. Protected routes require `Authorization: Bearer <token>` (else 401). orders (auth required): - POST /orders {"items":[{"sku","qty"}]} -> 201 {"order_id","total","status":"pending"} (total = sum of qty * a fixed per-sku price table you define). - GET /orders/{id} -> 200 the order, but OWNER-ONLY: a user may read only their OWN order; another user's order -> 403 (object-level authorization — do not infer access from the id alone). payments (auth required): - POST /payments {"order_id","amount"} -> 201 {"payment_id","status":"paid"} and flip the order's status to 'paid'. IDEMPOTENT on an `Idempotency-Key` header: the same key replays the same payment and must NOT double-charge or create a second payment row. Data MUST survive a process restart. Author the tests INDEPENDENTLY: unit for the password hash+verify and the idempotency-key dedup; integration for the owner-only 403 and the pay-flips-order-status flow; and prove the RUNNING service end-to-end (register -> login -> create order -> pay -> the order reads back 'paid', still 'paid' after a restart).

def _get_free_port() -> int:
    with contextlib.closing(socket.socket(socket.AF_INET, socket.SOCK_STREAM)) as s:
        s.bind(("127.0.0.1", 0))
        return int(s.getsockname()[1])


def _wait_for_port(host: str, port: int, *, timeout_s: float = 5.0) -> None:
    deadline = time.time() + timeout_s
    last_err: OSError | None = None
    while time.time() < deadline:
        try:
            with contextlib.closing(socket.create_connection((host, port), timeout=0.25)):
                return
        except OSError as e:  # pragma: no cover (timing)
            last_err = e
            time.sleep(0.05)
    raise RuntimeError(f"server did not start on {host}:{port}: {last_err}")


class ServerProcess:
    def __init__(self, *, port: int, db_path: str) -> None:
        self.port = port
        self.db_path = db_path
        self._proc: subprocess.Popen[str] | None = None

    def start(self) -> None:
        env = os.environ.copy()
        env["PORT"] = str(self.port)
        env["DB_PATH"] = self.db_path

        # Boot the real server entrypoint in a separate process.
        self._proc = subprocess.Popen(
            [sys.executable, "-u", "main.py"],
            env=env,
            stdout=subprocess.PIPE,
            stderr=subprocess.PIPE,
            text=True,
        )
        try:
            _wait_for_port("127.0.0.1", self.port, timeout_s=7.5)
        except Exception as err:
            # If the server failed fast, surface its stderr for debugging.
            if self._proc is not None:
                with contextlib.suppress(ProcessLookupError):
                    self._proc.terminate()
                out, stderr = self._proc.communicate(timeout=2)
                raise RuntimeError(
                    f"server failed to start. stdout=\n{out}\nstderr=\n{stderr}"
                ) from err
            raise

    def stop(self) -> None:
        if self._proc is None:
            return
        self._proc.terminate()
        try:
            self._proc.wait(timeout=5)
        except subprocess.TimeoutExpired:  # pragma: no cover
            self._proc.kill()
            self._proc.wait(timeout=5)

    def __enter__(self) -> "ServerProcess":
        self.start()
        return self

    def __exit__(self, exc_type: object, exc: object, tb: object) -> None:
        self.stop()


class HttpClient:
    def __init__(self, host: str, port: int) -> None:
        self.host = host
        self.port = port

    def request_json(
        self,
        method: str,
        path: str,
        *,
        body: dict[str, Any] | None = None,
        headers: dict[str, str] | None = None,
        timeout_s: float = 3.0,
    ) -> tuple[int, dict[str, Any]]:
        hdrs = {"Accept": "application/json"}
        if headers:
            hdrs.update(headers)
        raw_body = None
        if body is not None:
            raw_body = json.dumps(body).encode("utf-8")
            hdrs.setdefault("Content-Type", "application/json")

        conn = HTTPConnection(self.host, self.port, timeout=timeout_s)
        try:
            conn.request(method, path, body=raw_body, headers=hdrs)
            resp = conn.getresponse()
            raw = resp.read()
            data = json.loads(raw.decode("utf-8")) if raw else {}
            return int(resp.status), data
        finally:
            conn.close()

                    return {}

                except json.JSONDecodeError as err:

import os
import sqlite3
import tempfile
import unittest

from commerce_api.auth.repository import UserRepository
from commerce_api.auth.service import AuthService
from commerce_api.migrations.runner import run_migrations
from commerce_api.orders.repository import OrderRepository
from commerce_api.orders.service import OrderService
from commerce_api.payments.repository import PaymentRepository
from commerce_api.payments.service import PaymentService


class UnitServiceTests(unittest.TestCase):
    def setUp(self) -> None:
        fd, path = tempfile.mkstemp(suffix=".sqlite3")
        os.close(fd)
        self.db_path = path
        run_migrations(self.db_path)
        self.conn = sqlite3.connect(self.db_path)
        self.conn.row_factory = sqlite3.Row

    def tearDown(self) -> None:
        self.conn.close()
        os.unlink(self.db_path)

    def test_register_and_login_roundtrip(self) -> None:
        auth = AuthService(UserRepository(self.conn))
        user_id = auth.register(username="alice", password="pw")
        self.assertIsInstance(user_id, int)
        token = auth.login(username="alice", password="pw")
        self.assertTrue(token)
        authed = auth.authenticate(token)
        self.assertEqual(authed, user_id)

    def test_owner_only_order_get(self) -> None:
        auth = AuthService(UserRepository(self.conn))
        alice = auth.register(username="alice", password="pw")
        bob = auth.register(username="bob", password="pw")
        orders = OrderService(OrderRepository(self.conn))
        oid, _ = orders.create_order(user_id=alice, amount_cents=100)
        with self.assertRaises(PermissionError):
            orders.get_order_for_owner(order_id=oid, user_id=bob)

    def test_payment_idempotent(self) -> None:
        auth = AuthService(UserRepository(self.conn))
        alice = auth.register(username="alice", password="pw")
        orders_repo = OrderRepository(self.conn)
        orders = OrderService(orders_repo)
        oid, _ = orders.create_order(user_id=alice, amount_cents=100)
        pay = PaymentService(order_repo=orders_repo, payment_repo=PaymentRepository(self.conn))
        r1 = pay.pay_order(user_id=alice, order_id=oid, idempotency_key="k1")
        r2 = pay.pay_order(user_id=alice, order_id=oid, idempotency_key="k1")
        self.assertEqual(r1, r2)
        order = orders_repo.get_order(order_id=oid)
        assert order is not None
        self.assertEqual(str(order["status"]), "paid")


if __name__ == "__main__":
    unittest.main()

MIGRATIONS: list[tuple[int, str]] = [

        1,

        CREATE TABLE IF NOT EXISTS migrations (
            version INTEGER PRIMARY KEY,
            applied_at TEXT NOT NULL DEFAULT (datetime('now'))
        );

        CREATE TABLE IF NOT EXISTS users (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            username TEXT NOT NULL UNIQUE,
            password_hash BLOB NOT NULL,
            salt BLOB NOT NULL
        );

        CREATE TABLE IF NOT EXISTS tokens (
            token TEXT PRIMARY KEY,
            user_id INTEGER NOT NULL,
            created_at TEXT NOT NULL DEFAULT (datetime('now')),
            FOREIGN KEY(user_id) REFERENCES users(id) ON DELETE CASCADE
        );

        CREATE TABLE IF NOT EXISTS orders (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            owner_user_id INTEGER NOT NULL,
            amount_cents INTEGER NOT NULL,
            status TEXT NOT NULL,
            created_at TEXT NOT NULL DEFAULT (datetime('now')),
            FOREIGN KEY(owner_user_id) REFERENCES users(id) ON DELETE CASCADE
        );

        CREATE TABLE IF NOT EXISTS payment_idempotency (
            idempotency_key TEXT PRIMARY KEY,
            user_id INTEGER NOT NULL,
            order_id INTEGER NOT NULL,
            response_json TEXT NOT NULL,
            created_at TEXT NOT NULL DEFAULT (datetime('now'))
        );

        CREATE INDEX IF NOT EXISTS idx_orders_owner ON orders(owner_user_id);
        """,
    ),
]


def ensure_foreign_keys(conn: sqlite3.Connection) -> None:
    conn.execute("PRAGMA foreign_keys = ON")


def run_migrations(db_path: str) -> None:
    conn = sqlite3.connect(db_path)
    try:
        ensure_foreign_keys(conn)
        conn.execute(
            "CREATE TABLE IF NOT EXISTS migrations (version INTEGER PRIMARY KEY, applied_at TEXT NOT NULL DEFAULT (datetime('now')))"
        )
        applied = {row[0] for row in conn.execute("SELECT version FROM migrations").fetchall()}
        for version, sql in MIGRATIONS:
            if version in applied:
                continue
            with conn:
                conn.executescript(sql)
                conn.execute("INSERT INTO migrations(version) VALUES (?)", (version,))
    finally:
        conn.close()

from __future__ import annotations

import hashlib
import hmac
import secrets
import sqlite3

from commerce_api.auth.repository import UserRepository


class AuthService:
    def __init__(self, repo: UserRepository) -> None:
        self._repo = repo

    def register(self, *, username: str, password: str) -> int:
        if not username or not password:
            raise ValueError("missing_fields")
        salt = secrets.token_bytes(16)
        pwd_hash = _hash_password(password=password, salt=salt)
        try:
            return self._repo.create_user(username=username, password_hash=pwd_hash, salt=salt)
        except sqlite3.IntegrityError as err:
            raise ValueError("username_taken") from err

    def login(self, *, username: str, password: str) -> str:
        creds = self._repo.get_user_credentials(username=username)
        if creds is None:
            raise ValueError("invalid_credentials")
        user_id, stored_hash, salt = creds
        if not hmac.compare_digest(stored_hash, _hash_password(password=password, salt=salt)):
            raise ValueError("invalid_credentials")
        token = self._repo.new_token()
        self._repo.store_token(token=token, user_id=user_id)
        return token

    def authenticate(self, token: str) -> int | None:
        if not token:
            return None
        return self._repo.get_user_id_for_token(token=token)


def _hash_password(*, password: str, salt: bytes) -> bytes:
    return hashlib.pbkdf2_hmac("sha256", password.encode("utf-8"), salt, 120_000)

from __future__ import annotations

from typing import Any

from commerce_api.orders.repository import OrderRepository
from commerce_api.payments.repository import PaymentRepository


class PaymentService:
    def __init__(self, *, order_repo: OrderRepository, payment_repo: PaymentRepository) -> None:
        self._orders = order_repo
        self._payments = payment_repo

    def pay_order(
        self, *, user_id: int, order_id: int, idempotency_key: str | None
    ) -> dict[str, Any]:
        if order_id <= 0:
            raise ValueError("invalid_order_id")
        if not idempotency_key:
            raise ValueError("missing_idempotency_key")

        prior = self._payments.get_idempotent_response(idempotency_key=idempotency_key)
        if prior is not None:
            return prior

        order = self._orders.get_order(order_id=order_id)
        if order is None:
            raise ValueError("order_not_found")
        if int(order["owner_user_id"]) != user_id:
            raise PermissionError("not_owner")

        current_status = str(order["status"])
        if current_status != "paid":
            self._orders.set_status(order_id=order_id, status="paid")

        response = {"order_id": int(order_id), "status": "paid"}
        self._payments.store_idempotent_response(
            idempotency_key=idempotency_key,
            user_id=user_id,
            order_id=order_id,
            response=response,
        )
        return response

from __future__ import annotations

import secrets
import sqlite3


class UserRepository:
    def __init__(self, conn: sqlite3.Connection) -> None:
        self._conn = conn

    def create_user(self, *, username: str, password_hash: bytes, salt: bytes) -> int:
        cur = self._conn.execute(
            "INSERT INTO users(username, password_hash, salt) VALUES (?, ?, ?)",
            (username, password_hash, salt),
        )
        self._conn.commit()
        if cur.lastrowid is None:
            raise RuntimeError("failed_to_create_user")
        return int(cur.lastrowid)

    def get_user_credentials(self, *, username: str) -> tuple[int, bytes, bytes] | None:
        row = self._conn.execute(
            "SELECT id, password_hash, salt FROM users WHERE username = ?",
            (username,),
        ).fetchone()
        if row is None:
            return None
        return int(row[0]), bytes(row[1]), bytes(row[2])

    def store_token(self, *, token: str, user_id: int) -> None:
        self._conn.execute("INSERT INTO tokens(token, user_id) VALUES (?, ?)", (token, user_id))
        self._conn.commit()

    def get_user_id_for_token(self, *, token: str) -> int | None:
        row = self._conn.execute("SELECT user_id FROM tokens WHERE token = ?", (token,)).fetchone()
        if row is None:
            return None
        return int(row[0])

    def new_token(self) -> str:
        return secrets.token_urlsafe(32)

from __future__ import annotations

import sqlite3


class OrderRepository:
    def __init__(self, conn: sqlite3.Connection) -> None:
        self._conn = conn

    def create_order(self, *, owner_user_id: int, amount_cents: int) -> int:
        cur = self._conn.execute(
            "INSERT INTO orders(owner_user_id, amount_cents, status) VALUES (?, ?, 'created')",
            (owner_user_id, amount_cents),
        )
        self._conn.commit()
        if cur.lastrowid is None:
            raise RuntimeError("failed_to_create_order")
        return int(cur.lastrowid)

    def get_order(self, *, order_id: int) -> sqlite3.Row | None:
        row = self._conn.execute(
            "SELECT id, owner_user_id, amount_cents, status FROM orders WHERE id = ?",
            (order_id,),
        ).fetchone()
        if row is None:
            return None
        if not isinstance(row, sqlite3.Row):
            raise TypeError("unexpected_row_type")
        return row

    def set_status(self, *, order_id: int, status: str) -> None:
        self._conn.execute("UPDATE orders SET status = ? WHERE id = ?", (status, order_id))
        self._conn.commit()
